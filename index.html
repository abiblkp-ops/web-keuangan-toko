async function processImageWithAI(event) { const file = event.target.files[0]; if (!file) return; const statusEl = document.getElementById('ocrStatus'); statusEl.innerText = "⏳ Mengirim nota ke AI..."; const reader = new FileReader(); reader.readAsDataURL(file); reader.onload = async function () { const base64Image = reader.result; try { // Kirim ke backend Anda sendiri yang memegang API Key OpenAI const response = await fetch('/api/parse-nota', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: { image: base64Image } }); const data = await response.json(); // Render data JSON dari AI ke tabel HTML Anda renderTableFromJSON(data.items); statusEl.innerText = "✅ Berhasil diproses oleh AI!"; } catch (err) { statusEl.innerText = "Gagal memproses dengan AI."; } }; }
